# يحتوي هذا الملف على رسائل الخطأ التي تظهر للمستخدم عند استخدام تسجيل الدخول عبر http/https[cite: 1].
# يمكن تغيير هذه الرسائل لجعل واجهة المستخدم أكثر سهولة، بما في ذلك الترجمة إلى لغات مختلفة[cite: 2].
#
# يمكن استخدام متغيرات مختلفة هنا أيضاً، وأكثرها شيوعاً هي:
#	$(error-orig)	- رسالة الخطأ الأصلية من الهوتسبوت 
#	$(ip)		- عنوان IP الخاص بالعميل 
#	$(username)	- اسم المستخدم الذي يحاول تسجيل الدخول 

# خطأ داخلي (internal-error)
# لا ينبغي أن يحدث أبداً، وفي حال حدوثه ستظهر صفحة الخطأ تعرض هذه الرسالة[cite: 4].
internal-error = خطأ داخلي ($(error-orig))

# خطأ في الإعدادات (config-error)
# لا يحدث إذا تم تكوين الهوتسبوت بشكل صحيح[cite: 5].
config-error = خطأ في التهيئة ($(error-orig)) [cite: 5]

# لم يتم تسجيل الدخول (not-logged-in)
# يحدث إذا طلب المستخدم صفحة الحالة أو تسجيل الخروج وهو غير مسجل دخول أصلاً.
not-logged-in = لم تقم بتسجيل الدخول (عنوان IP: $(ip))

# نفاذ عناوين IP (ippool-empty)
# عندما لا تتوفر عناوين IP في "Pool" المخصص للمستخدمين.
ippool-empty = لا يمكن تخصيص عنوان IP - لا توجد عناوين متاحة في التجمع

# إيقاف الخدمة (shutting-down)
# عند تنفيذ أمر إيقاف التشغيل، لا يتم قبول عملاء جدد.
shutting-down = خدمة الهوتسبوت قيد الإيقاف

# حد الجلسات (user-session-limit)
# إذا كان بروفايل المستخدم لديه حد للمستخدمين المشتركين (shared-users)[cite: 6].
user-session-limit = لا يُسمح بمزيد من الجلسات للمستخدم $(username) [cite: 6]

# حد ترخيص النظام (license-session-limit)
# يعتمد على رقم رخصة الميكروتيك لعدد مستخدمي الهوتسبوت النشطين[cite: 7].
license-session-limit = تم الوصول إلى حد الجلسات المسموح به للنظام ($(error-orig)) [cite: 7]

# ماك أدرس خاطئ (wrong-mac-username)
# إذا كان اسم المستخدم يشبه عنوان MAC ولكنه لا يطابق عنوان العميل الحقيقي.
wrong-mac-username = اسم المستخدم غير صالح ($(username)): عنوان MAC هذا ليس ملكك

# فقدان التشفير (chap-missing)
# يظهر عند استخدام طريقة http-chap وفشل المتصفح في إرسال كلمة المرور المشفرة[cite: 8].
# الأسباب المحتملة: جافا سكريبت معطلة، صفحة login.html غير صالحة، أو انتهاء صلاحية التحدي[cite: 8, 9, 10].
chap-missing = لم يرسل المتصفح استجابة التشفير (جرب مرة أخرى، أو فعّل JavaScript في متصفحك) [cite: 11]

# اسم مستخدم غير صالح (invalid-username)
# الحالة العامة لخطأ اسم المستخدم أو كلمة المرور[cite: 12].
# ملاحظة: إذا أرسل سيرفر RADIUS رسالة رفض (Access-Reject) نصية، فستحل محل هذه الرسالة[cite: 12].
invalid-username = اسم المستخدم أو كلمة المرور غير صحيحة [cite: 13]

# عنوان ماك غير مسموح (invalid-mac)
# عندما يكون المستخدم مرتبطاً بعنوان MAC معين ويحاول الدخول من جهاز آخر[cite: 14, 15].
invalid-mac = المستخدم $(username) غير مسموح له بتسجيل الدخول من عنوان MAC هذا (يرجى مراجعة الإدارة) [cite: 15]

# حدود الوقت والبيانات (uptime-limit, traffic-limit)
uptime-limit = وصل المستخدم $(username) إلى حد زمن الاتصال المسموح به
traffic-limit = وصل المستخدم $(username) إلى حد حجم البيانات (التحميل) المسموح به

# مهلة سيرفر الراديوس (radius-timeout)
# عندما لا يتلقى الراوتر رداً من خادم RADIUS[cite: 16].
radius-timeout = خادم RADIUS لا يستجيب، يرجى المحاولة لاحقاً [cite: 16]

# عملية المصادقة قيد التنفيذ (auth-in-progress)
# العميل لديه طلب مصادقة قيد المعالجة حالياً[cite: 17].
auth-in-progress = جاري المصادقة، يرجى الانتظار قليلاً [cite: 18]

# رسالة رد الراديوس (radius-reply)
# عندما يرسل سيرفر الراديوس رسالة خطأ مخصصة.
radius-reply = رسالة من النظام: $(error-orig)
